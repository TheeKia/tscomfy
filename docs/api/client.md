# @tscomfy/client: API surface

Generated by `bun run api:update` from the package's declarations: every `.d.ts` file a
consumer's type checker reads, starting at the `types` entry. Don't edit it by hand;
`bun run check` fails while it is stale. A change here is a change to the public API.

## `index.d.ts`

```ts
/**
 * @module @tscomfy/client
 *
 * Runs tscomfy workflows on a ComfyUI server: `connect` opens a websocket, `submit` queues a
 * workflow and returns a job whose events stream progress, live previews and outputs, and whose
 * outcome the server's record of the job decides. `uploadFile` and `fetchFile` move files.
 * Runtime-agnostic: it uses only `fetch`, `WebSocket` and their companions.
 */
export { ClientError } from "./client-error.ts";
```

## `client-error.d.ts`

```ts
/**
 * The server couldn't be reached, or answered in a way the protocol doesn't allow. Thrown by the
 * HTTP functions and used to reject a job's outcome; what the server decides about a job (a
 * rejection, a failure) is data instead, in the {@link Outcome}. The underlying error, if any, is
 * the `cause`.
 */
export declare class ClientError extends Error {
    /** The HTTP status of the answer that caused it, when there was one. */
    readonly status: number | undefined;
    constructor(message: string, options?: {
        readonly cause?: unknown;
        readonly status?: number;
    });
}
```

## `connection.d.ts`

```ts
import type { JsonObject } from "@tscomfy/core";
import type { Server } from "./server.ts";
/**
 * An open connection to a server: a {@link Server} with a websocket that delivers the events of
 * the jobs submitted through it. Several jobs share it. Close it when done: an open websocket
 * keeps a Node or Bun process running.
 */
export interface Connection extends Server {
    /**
     * The id the server sends this connection's events to. A new one for every websocket, so it
     * changes when the connection reconnects.
     */
    readonly clientId: string;
    /** The server's feature flags, as it answered them on the current websocket. */
    readonly features: JsonObject;
    /**
     * Closes the websocket. Jobs still open reject their outcome with a {@link ClientError}; they
     * keep running on the server.
     */
    close(): Promise<void>;
}
/** Options for {@link connect}. */
export interface ConnectOptions {
    /** Aborts opening the connection; rejects with the signal's reason. */
    readonly signal?: AbortSignal;
}
/**
 * Opens a connection: a websocket with a fresh client id, which announces
 * `supports_preview_metadata` (so previews say which job and node they belong to) and waits for
 * the server's feature flags. Also checks that the server is ComfyUI 0.26 or later with the job
 * API. Rejects with a {@link ClientError} when the server can't be reached or is too old, and
 * with the signal's reason when `signal` aborts; either way nothing is left open. Throws a
 * `TypeError` for an invalid `server.url`.
 */
export declare function connect(server: Server, options?: ConnectOptions): Promise<Connection>;
```

## `events.d.ts`

```ts
import type { NodeId } from "@tscomfy/core";
import type { NodeOutput } from "./outputs.ts";
/** The server started running the job. */
export interface StartedEvent {
    readonly type: "started";
}
/** Nodes whose cached results the run reuses instead of running them. */
export interface CachedEvent {
    readonly type: "cached";
    readonly nodes: readonly NodeId[];
}
/** A node started running. */
export interface ExecutingEvent {
    readonly type: "executing";
    readonly node: NodeId;
}
/** A running node's progress: `value` of `max` (sampler steps, say). */
export interface ProgressEvent {
    readonly type: "progress";
    readonly node: NodeId;
    readonly value: number;
    readonly max: number;
}
/**
 * A live preview from a running node: a sampler's latent preview (a JPEG at most 512 pixels on a
 * side, sent when the job asks for previews with `previews` in {@link SubmitOptions}, or the
 * server was started with `--preview-method`), or a `SaveImageWebsocket` image (a full-size PNG,
 * always sent).
 */
export interface PreviewEvent {
    readonly type: "preview";
    readonly node: NodeId;
    /** `image/jpeg` or `image/png`. */
    readonly mime: string;
    readonly bytes: Uint8Array<ArrayBuffer>;
}
/** A node finished and reported output (files, text). */
export interface OutputEvent {
    readonly type: "output";
    readonly node: NodeId;
    readonly output: NodeOutput;
}
/** Progress text from a running node (partner nodes report their status this way). */
export interface TextEvent {
    readonly type: "text";
    readonly node: NodeId;
    readonly text: string;
}
/**
 * A node blocked execution with a message. The run continues without what depends on it; the
 * outcome also lists it, as a `W_NODE_BLOCKED` diagnostic.
 */
export interface BlockedEvent {
    readonly type: "blocked";
    readonly node: NodeId;
    readonly message: string;
}
/** What a job reports while it runs. Node ids are the prompt's (the builder's) ids. */
export type JobEvent = StartedEvent | CachedEvent | ExecutingEvent | ProgressEvent | PreviewEvent | OutputEvent | TextEvent | BlockedEvent;
```

## `files.d.ts`

```ts
import type { FileRef, FolderType } from "./outputs.ts";
import type { Server } from "./server.ts";
/** A file to upload. */
export interface UploadSource {
    /** The file name to store it under. */
    readonly name: string;
    readonly bytes: Uint8Array;
    /** Its media type, such as `image/png`. */
    readonly mime?: string;
}
/** Options for {@link uploadFile}. */
export interface UploadOptions {
    /** A folder under the target directory. */
    readonly subfolder?: string;
    /** The directory: `input` (default), `temp` or `output`. */
    readonly type?: FolderType;
    /** Replace a file of the same name. Otherwise the server keeps both, renaming the new one. */
    readonly overwrite?: boolean;
    readonly signal?: AbortSignal;
}
/** An uploaded file, as the server stored it. */
export interface UploadedFile extends FileRef {
    /**
     * The value a loader node's file input takes (`LoadImage`'s `image`): `subfolder/name`, with
     * ` [temp]` or ` [output]` after it outside the input directory.
     */
    readonly value: string;
}
/**
 * Uploads a file (an image, video or audio for a loader node) through `/upload/image`. The
 * stored name can differ from `source.name`: without `overwrite`, the server renames a file whose
 * name is taken by different content. Rejects with a {@link ClientError} if the server refuses.
 */
export declare function uploadFile(server: Server, source: UploadSource, options?: UploadOptions): Promise<UploadedFile>;
/** A downloaded file. */
export interface FileData {
    readonly bytes: Uint8Array<ArrayBuffer>;
    /** The server's `Content-Type`; `application/octet-stream` when it sent none. */
    readonly mime: string;
}
/** Options for {@link fetchFile}. */
export interface FetchFileOptions {
    readonly signal?: AbortSignal;
}
/**
 * Downloads a file the server has, such as an output, through `/view`. Rejects with a
 * {@link ClientError} if the server doesn't have it.
 */
export declare function fetchFile(server: Server, file: FileRef, options?: FetchFileOptions): Promise<FileData>;
/**
 * The URL `/view` serves a file at, for `<img src>` on a page the server allows (same origin, or
 * `--enable-cors-header`). It carries the server URL's query, so a token in it is in the result.
 */
export declare function fileUrl(server: Server, file: FileRef): string;
```

## `job.d.ts`

```ts
import { type ApiPrompt, type JsonObject, type Workflow } from "@tscomfy/core";
import type { Connection } from "./connection.ts";
import type { JobEvent } from "./events.ts";
import type { Outcome } from "./outcome.ts";
import type { NodeRef } from "./outputs.ts";
/** How the server draws live previews: `auto` is latent2rgb; `taesd` needs its decoder model. */
export type PreviewMethod = "auto" | "latent2rgb" | "taesd" | "none";
/** Options for {@link submit}. */
export interface SubmitOptions {
    /**
     * Live previews for this job (`extra_data.preview_method`). Without it the server's default
     * applies, which is `none` unless it was started with `--preview-method`.
     */
    readonly previews?: PreviewMethod;
    /** Run only these output nodes (`partial_execution_targets`). */
    readonly targets?: readonly NodeRef[];
    /** Queue ahead of the jobs already waiting. */
    readonly front?: boolean;
    /**
     * More `extra_data` for the server, such as `api_key_comfy_org` for partner nodes. The keys the
     * client sets (`client_id`, `preview_method`) are refused.
     */
    readonly extraData?: JsonObject;
    /**
     * Aborting it cancels the job on the server and rejects the outcome with the signal's reason:
     * `AbortSignal.timeout(600_000)`. There is no default timeout.
     */
    readonly signal?: AbortSignal;
}
/** A submitted job. */
export interface Job {
    /** The prompt id, minted by the client. */
    readonly id: string;
    /**
     * What the job reports while it runs, for one consumer, from the moment it was submitted. Ends
     * when the outcome settles; never throws for the job's failure.
     */
    readonly events: AsyncIterable<JobEvent>;
    /**
     * How the job ended, as the server's record says. Rejects only with the signal's reason (see
     * {@link SubmitOptions.signal}) or a {@link ClientError}: the server couldn't be reached, the
     * connection was closed, or the server broke the protocol.
     */
    readonly outcome: Promise<Outcome>;
    /**
     * Asks the server to cancel the job, queued or running. The outcome then settles, normally as
     * `cancelled` (a job that was finishing may still succeed). Rejects with a {@link ClientError}
     * if the server can't be asked.
     */
    cancel(): Promise<void>;
}
/**
 * Submits a workflow, or a prompt, to run on the server. A workflow is exported with `toApi`,
 * which validates it and throws a `ValidationError` for errors (D21); a prompt is sent as given.
 * The job is registered under a prompt id the client mints before anything is sent, so no event
 * is missed. Throws a `TypeError` for invalid options or a closed connection.
 */
export declare function submit(connection: Connection, input: Workflow | ApiPrompt, options?: SubmitOptions): Job;
```

## `outcome.d.ts`

```ts
import type { Diagnostic, NodeId } from "@tscomfy/core";
import type { Outputs } from "./outputs.ts";
/** The job ran to the end. Outputs left out by a partial acceptance are in `diagnostics`. */
export interface SuccessOutcome {
    readonly status: "success";
    readonly outputs: Outputs;
    /**
     * `E_SERVER_VALIDATION` for outputs the server refused while running the others, and
     * `W_NODE_BLOCKED` for nodes that blocked execution.
     */
    readonly diagnostics: readonly Diagnostic[];
}
/** The server refused the prompt: nothing ran. */
export interface RejectedOutcome {
    readonly status: "rejected";
    /** `E_PROMPT_REJECTED` for the server's error, and `E_SERVER_VALIDATION` for each node error. */
    readonly diagnostics: readonly Diagnostic[];
}
/** A node raised an error; the run stopped there. */
export interface FailedOutcome {
    readonly status: "failed";
    readonly error: ExecutionError;
    /** What the nodes that finished before the error reported. */
    readonly outputs: Outputs;
    readonly diagnostics: readonly Diagnostic[];
}
/** The job was cancelled or interrupted, by this client or another. */
export interface CancelledOutcome {
    readonly status: "cancelled";
    /** What the nodes that finished before the cancel reported. */
    readonly outputs: Outputs;
    readonly diagnostics: readonly Diagnostic[];
}
/** How a job ended, as the server's record of it says. */
export type Outcome = SuccessOutcome | RejectedOutcome | FailedOutcome | CancelledOutcome;
/** The error a node raised. */
export interface ExecutionError {
    readonly node: NodeId;
    readonly classType: string;
    /** The exception's class, qualified: `torch.OutOfMemoryError`, `ValueError`. */
    readonly exceptionType: string;
    readonly message: string;
    readonly traceback: readonly string[];
    /** The nodes that had finished. */
    readonly executed: readonly NodeId[];
}
```

## `outputs.d.ts`

```ts
import type { NodeHandle, NodeId, JsonObject, Out } from "@tscomfy/core";
import type { Outcome } from "./outcome.ts";
/** Which of the server's folders a file is in. */
export type FolderType = "output" | "temp" | "input";
/** A file on the server, as outputs name it and `/view` serves it. */
export interface FileRef {
    readonly filename: string;
    /** The folder under {@link FileRef.type}'s directory; `""` for none. */
    readonly subfolder: string;
    readonly type: FolderType;
}
/** What a file holds, from the server's metadata, the file's extension, or its output key. */
export type FileKind = "image" | "video" | "audio" | "3d" | "file";
/** A file a node produced. */
export interface OutputFile extends FileRef {
    readonly kind: FileKind;
    /**
     * The node marked its images animated. Animated WEBP and PNG stay `image`; core's video nodes
     * report their files under `images` with this flag, and their kind is `video`.
     */
    readonly animated: boolean;
    /** The key of the node's UI output it was listed under: `images`, `audio`, `3d`, `gifs`, … */
    readonly key: string;
    /** Pixels, when the server measured the file. */
    readonly width?: number;
    readonly height?: number;
}
/** What one node reported when it ran: its files, its text, and the raw record. */
export interface NodeOutput {
    /** The node, by its id in the prompt (the builder's id for tscomfy workflows). */
    readonly node: NodeId;
    /** Every file entry under any key, in the order the node listed them. */
    readonly files: readonly OutputFile[];
    /** The strings under `text` (PreviewAny, text previews). */
    readonly text: readonly string[];
    /** The node's UI output as the server sent it: lists by key. */
    readonly ui: JsonObject;
}
/** Every node's output, by node id, in prompt order. */
export type Outputs = ReadonlyMap<NodeId, NodeOutput>;
/** A node: its id, its handle (`n.SaveImage(…)`), or one of its outputs. */
export type NodeRef = NodeId | NodeHandle | Out<string>;
/**
 * What a node (by id, handle or output) reported, or `undefined` when it reported nothing, didn't
 * run, or the job was rejected.
 */
export declare function outputOf(outcome: Outcome, node: NodeRef): NodeOutput | undefined;
/**
 * The files of the given nodes (by id, handle or output), or of every node when none is given, in
 * prompt order. Empty for a rejected job, which has no outputs.
 */
export declare function outputFiles(outcome: Outcome, ...nodes: readonly NodeRef[]): readonly OutputFile[];
```

## `server.d.ts`

```ts
/**
 * A ComfyUI server: where it is and how to talk to it. Plain data; every function that needs no
 * websocket takes one, and a {@link Connection} is one too.
 */
export interface Server {
    /**
     * The server's http(s) URL. It may have a path prefix and a query (a token, say), which every
     * request and the websocket URL keep: `http://host/comfy/?token=t` gives
     * `http://host/comfy/prompt?token=t`. Credentials (`user:password@`) are refused, because
     * runtimes disagree on them; send them in `headers`.
     */
    readonly url: string;
    /**
     * Headers for every HTTP request, such as `Authorization`. Not sent on the websocket
     * handshake, which browsers can't give headers; see {@link Server.webSocket}.
     */
    readonly headers?: Readonly<Record<string, string>>;
    /**
     * Replaces the global `fetch`: the global itself, or a function wrapping it (to add a proxy, a
     * cookie, logging). It is called with a URL and an init.
     */
    readonly fetch?: (url: URL, init: RequestInit) => Promise<Response>;
    /**
     * Opens the websocket instead of `new WebSocket(url)`. Bun, Node and Deno accept handshake
     * headers this way: `(url) => new WebSocket(url, { headers })` (not standard, so not in the
     * types; browsers throw).
     */
    readonly webSocket?: (url: string) => WebSocket;
}
```
