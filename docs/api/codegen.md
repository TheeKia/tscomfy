# @tscomfy/codegen: API surface

Generated by `bun run api:update` from the package's declarations: every `.d.ts` file a
consumer's type checker reads, starting at the `types` entry. Don't edit it by hand;
`bun run check` fails while it is stale. A change here is a change to the public API.

## `generate.d.ts`

```ts
import { type Diagnostic, type Schema } from "@tscomfy/core";
/** The version written into generated headers. A test keeps it equal to package.json. */
export declare const CODEGEN_VERSION = "0.1.0";
/** A file to write, as data: codegen never touches the filesystem. */
export interface GeneratedFile {
    /** Relative to the output directory. */
    readonly path: string;
    readonly contents: string;
}
/** Options for {@link generate}. */
export interface GenerateOptions {
    /** The exported schema constant, e.g. `local`: one `exportNameProblem` finds nothing wrong with. */
    readonly exportName: string;
    /**
     * The extension `index.ts` uses to import `nodes.ts`: `.js` (the default) works under
     * `nodenext` and `bundler` resolution; `.ts` needs `allowImportingTsExtensions` or
     * `rewriteRelativeImportExtensions`.
     */
    readonly importExtension?: ".js" | ".ts";
    /** The command that regenerates the files, recorded in their headers. */
    readonly command?: string;
}
/** The result of {@link generate}. */
export interface GenerateResult {
    /** `nodes.ts` and `index.ts`. */
    readonly files: readonly GeneratedFile[];
    readonly diagnostics: readonly Diagnostic[];
}
/**
 * Generates the files for one schema: `nodes.ts` (types only) and `index.ts` (the typed runtime
 * schema). Pure and deterministic. Throws a `TypeError` for an export name `exportNameProblem`
 * refuses.
 */
export declare function generate(schema: Schema, options: GenerateOptions): GenerateResult;
```

## `export-name.d.ts`

```ts
/** What the schema module (`index.ts`) imports. */
export declare const SCHEMA_MODULE_NAMES: readonly string[];
/** What a transpiled module imports or binds itself. */
export declare const TRANSPILED_MODULE_NAMES: readonly string[];
/**
 * Why `name` can't be the exported constant of a generated module, or `undefined` when it can.
 * One rule for both kinds: a schema module's export (`generate`) and a transpiled workflow's
 * (`transpile`), which imports the schema's, so a schema named `raw` is refused when it is
 * generated rather than when a workflow is first transpiled over it. The name must be an ASCII
 * identifier, not a reserved word, and none of the names either module declares itself
 * (`defineSchema`, `TypedSchema`, `Nodes`, `workflow`, `Workflow`, `NodesOf`, `n`, `raw`).
 */
export declare function exportNameProblem(name: string): string | undefined;
/** Whether {@link exportNameProblem} finds nothing wrong with `name`. */
export declare function isExportName(name: string): boolean;
```

## `transpile.d.ts`

```ts
import { type Diagnostic, type Workflow } from "@tscomfy/core";
/** Options for {@link transpile}. */
export interface TranspileOptions {
    /** The exported constant holding the workflow, e.g. `txt2img`. */
    readonly name: string;
    /** Where the generated schema is imported from, e.g. `./comfy/local/index.js`. */
    readonly schemaModule: string;
    /** The schema module's export, e.g. `local`. */
    readonly schemaName: string;
    /** Inline a node used once into its use, unless that node inlines another. Default `true`. */
    readonly inline?: boolean;
    /** Extra header lines, e.g. the command that transpiled the file. */
    readonly header?: readonly string[];
}
/** What {@link transpile} returns. */
export interface TranspileResult {
    /** The module source, formatted as oxfmt formats it. */
    readonly code: string;
    readonly diagnostics: readonly Diagnostic[];
}
/**
 * Builder code for `workflow`: an ES module exporting the workflow, built with the generated types
 * over `options.schemaName`. Nodes keep their ids and titles and come in dependency order; inputs
 * equal to their default are left out, since the exporter fills them in. Running it and exporting
 * gives the same prompt as `toApi(workflow)`, except that ids that aren't integers (`40:31`) come
 * in dependency order, where the workflow lists one before a node it depends on. Pure and
 * deterministic.
 *
 * Reports `W_TRANSPILED_AS_RAW` for a node the types can't express, which is written as `raw(…)`,
 * and `E_DEPENDENCY_CYCLE` and `E_DANGLING_LINK` for what no program can build: those nodes, or
 * links, are left out.
 */
export declare function transpile(workflow: Workflow, options: TranspileOptions): TranspileResult;
/**
 * Why {@link transpile} refuses these names, or `undefined` when it takes them: each must pass
 * `exportNameProblem`, they must differ, and the declaration
 * `export const <name>: Workflow<NodesOf<typeof <schemaName>>> =` must fit on one line.
 */
export declare function transpileNameProblem(options: Pick<TranspileOptions, "name" | "schemaName">): string | undefined;
```
