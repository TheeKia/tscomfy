```ts
import type { Workflow } from "../graph/graph.ts";
import type { ApiPrompt } from "./prompt.ts";
/** Options for {@link toApi} and {@link exportApi}. */
export interface ToApiOptions {
    /** Emit `_meta.title`, as the frontend does. Default `true`. */
    readonly meta?: boolean;
    /**
     * Validate the prompt against the schema, as {@link validateApi} does. Default `true`.
     * Problems building the prompt itself are reported either way.
     */
    readonly validate?: boolean;
    /**
     * Apply bypass and mute. Default `true`. With `false`, every node is sent as if it ran, as a tool
     * that rewrites workflows needs (the transpiler).
     */
    readonly modes?: boolean;
}
/** What {@link exportApi} returns. */
export interface ApiExportResult {
    /** The prompt, built even when there are errors: whatever could be sent. */
    readonly prompt: ApiPrompt;
    /** Problems building the prompt, then, unless `validate` is `false`, validating it. */
    readonly diagnostics: readonly Diagnostic[];
}
/**
 * Converts a workflow into an API-format prompt: explicit schema defaults filled in, dynamic inputs
 * flattened to their wire keys, list values wrapped, keys in the frontend's order. Serialize it
 * with `stringify`, which keeps big integers exact.
 *
 * Throws a {@link ValidationError}, carrying every diagnostic, when there is an error: the prompt
 * can't be built faithfully (`E_INPUT_KEY_COLLISION`, `E_TOO_MANY_DYNAMIC_INPUTS`), or, unless
 * `validate` is `false`, the backend would reject it. Warnings don't throw; use {@link exportApi}
 * to see them. Throws a `TypeError` for an output reference nested inside a JSON value, which no
 * input can carry.
 */
export declare function toApi(workflow: Workflow, options?: ToApiOptions): ApiPrompt;
/**
 * {@link toApi} without throwing for diagnostics: the prompt, and what went wrong building and
 * validating it. Still throws a `TypeError` where {@link toApi} does.
 */
export declare function exportApi(workflow: Workflow, options?: ToApiOptions): ApiExportResult;
/**
 * Every problem with a workflow: the errors {@link toApi} would throw for, and the warnings, which
 * don't stop the backend (a value outside an upload combo's listed files, for example). Empty for a
 * workflow the backend accepts with nothing to report. Throws a `TypeError` where {@link toApi}
 * does.
 */
export declare function validate(workflow: Workflow): readonly Diagnostic[];
```

## `graph/types.d.ts`

```ts
/** A node's id in a prompt: `"1"`, `"2"`, … unless set explicitly. */
export type NodeId = string;
declare const socketType: unique symbol;
/**
 * A reference to output `slot` of node `node`, carrying the socket type `T` it produces. Branded,
 * so a plain `[id, slot]` tuple or an object of the right shape can't pass for one, and
 * covariant, so `Out<"MODEL">` fits wherever `Out<"MODEL" | "*">` is expected.
 */
export interface Out<out T extends string> {
    readonly [socketType]: T;
    readonly node: NodeId;
    readonly slot: number;
}
/**
 * What a socket of type `T` accepts: an output of type `T`, or an any-typed (`*`) output. The
 * backend accepts `*` on either side.
 */
export type In<T extends string> = Out<T | "*">;
/**
 * The link type of a MatchType input after the first one of its template: the inferred
 * type or `*`, or any output when the template was inferred as `*`.
 */
export type Match<T extends string> = "*" extends T ? string : T | "*";
/**
 * The socket type tscomfy gives an output that is a list of combo options. The backend
 * connects one only into an input typed exactly `COMBO` or `*`, so it needs a name of its own:
 * a plain `COMBO` output also fits comma unions that contain `COMBO`.
 */
export type OptionList = "COMBO (option list)";
/**
 * A list type of a limited length: `T` with `length` one of `L`. Generated types use it
 * for Autogrow slots and DynamicGroup rows, whose count the backend bounds.
 */
export type Bounded<T extends readonly unknown[], L extends number> = T & {
    readonly length: L;
};
/** What a node without outputs returns. */
export interface NodeHandle {
    readonly id: NodeId;
}
/** Per-node options, the second argument of every node function. */
export interface NodeOptions {
    /** An explicit id instead of the next sequential one. */
    readonly id?: NodeId;
    /** Sent as `_meta.title`. */
    readonly title?: string;
    /**
     * `"bypass"`: the node isn't sent, and links through it reroute to one of its inputs, as the
     * frontend's bypass does. `"mute"`: the node isn't sent. An input whose link then resolves to
     * nothing is left out, or, for a widget, sends the widget's value instead.
     */
    readonly mode?: NodeMode;
}
/** How a node takes part in a prompt, besides normally. */
export type NodeMode = "bypass" | "mute";
export {};
```

```ts
/** The result of {@link parse}. */
export interface ParseResult {
    /** The parsed value, or `undefined` when the text is not valid JSON. */
    readonly value: JsonValue | undefined;
    /** `E_INVALID_JSON`, `W_NON_FINITE_NUMBER` or `W_PRECISION_LOSS`; see {@link parse}. */
    readonly diagnostics: readonly Diagnostic[];
}
/**
 * Parses JSON text, keeping integers beyond 2^53 exact as `bigint` wherever they occur, however
 * long. Integer literals inside the safe range, and every literal with a fraction or an exponent,
 * stay `number`. The result never holds a non-finite number.
 *
 * Never throws for bad input:
 * - invalid JSON yields `E_INVALID_JSON` and no value;
 * - `NaN`, `Infinity` and `-Infinity` (which Python servers write for non-finite floats), and
 *   literals with a fraction or an exponent beyond the double range (`1e400`), become `null` with
 *   `W_NON_FINITE_NUMBER`;
 * - on runtimes without `JSON.parse` source-text access, each integer beyond 2^53 may have lost
 *   precision and yields `W_PRECISION_LOSS`, and one beyond the double range becomes `null` with
 *   `W_NON_FINITE_NUMBER`.
 */
export declare function parse(text: string): ParseResult;
/**
 * {@link parse} with source-text access forced on or off. Exported for tests of the fallback;
 * not part of the public API.
 */
export declare function parseWith(text: string, sourceAccess: boolean): ParseResult;
```

## `json/stringify.d.ts`

```ts
import { type JsonValue } from "./value.ts";
/**
 * Serializes a JSON value as compact JSON: keys in JavaScript's enumeration order (integer-like
 * keys first in numeric order, then insertion order, as `JSON.stringify` and the ComfyUI frontend
 * use), `bigint` written as a plain integer literal, everything else exactly as `JSON.stringify`
 * writes it, with one exception: an integer-valued `number` beyond 2^53 is written in exponent form
 * (`1.844674407370955e+19`). `JSON.stringify` would print it as an integer literal, which `parse`
 * reads back as a `bigint` of a different value; in exponent form it round-trips as the same
 * `number`, and Python reads it as a float too. The output for a given value is the same in every
 * runtime.
 *
 * Throws a `TypeError` for what has no JSON form: a non-finite number, and `undefined` (a member
 * set to it, an array hole), which the types rule out but code without `exactOptionalPropertyTypes`
 * can still pass. Passing one is a programmer error.
 */
export declare function stringify(value: JsonValue): string;
```

```ts
declare const nodesType: unique symbol;
/**
 * A schema together with the generated `Nodes` interface that types its node functions. The
 * brand exists only at compile time; {@link defineSchema} and {@link typedSchema} apply it.
 */
export type TypedSchema<N> = Schema & {
    readonly [nodesType]: N;
};
/**
 * The `Nodes` interface of a typed schema: `Workflow<NodesOf<typeof local>>` names a workflow's
 * type without importing the generated `nodes.ts`.
 */
export type NodesOf<S> = S extends TypedSchema<infer N> ? N : never;
/**
 * Loads the schema a generated `index.ts` embeds. `text` is `stringify(schema)` as codegen wrote
 * it; it is parsed losslessly, so big integers stay exact.
 *
 * Throws when the text isn't such a schema: generated files are tscomfy's own output, so a
 * mismatch means the file was edited or truncated, which is misuse rather than data to report.
 * Also throws when the schema holds integers beyond 2^53 (every real one does: seed bounds) and
 * the runtime's `JSON.parse` has no source-text access to read them exactly.
 */
export declare function defineSchema<N>(text: string): TypedSchema<N>;
/**
 * A schema as a typed schema, to build and import workflows over a schema read at runtime (from
 * `normalizeObjectInfo`) rather than generated. With `N` left `unknown`, `n` has no node
 * functions, and nodes are added with `raw`; `importApi` and `importUi` work as with generated
 * types.
 *
 * The result is frozen. The node definitions aren't copied: they are frozen in place, as the
 * normalizer already returns them, so a schema built by hand can't be changed afterwards either.
 */
export declare function typedSchema<N = unknown>(schema: Schema): TypedSchema<N>;
/** {@link defineSchema} from a parse result. Exported for tests; not part of the public API. */
export declare function schemaFromParse<N>({ value, diagnostics }: ParseResult): TypedSchema<N>;
```

```ts
import type { JsonValue } from "../json/value.ts";
/**
 * An integer as the schema stores it: a `number` when it is a safe integer, otherwise a `bigint`.
 * One value never has both forms.
 */
export type IntValue = number | bigint;
/**
 * A combo option. Servers also use numbers and booleans, and the backend compares values exactly,
 * so they are kept as they are.
 */
export type ComboOption = string | number | boolean;
/**
 * The type of a socket, parsed from a ComfyUI type string. Names are trimmed, as the backend
 * trims them when it compares.
 */
export type SocketType = 
/** One type, e.g. `MODEL`. */
{
    readonly kind: "named";
    readonly name: string;
}
/** A comma list such as `IMAGE,MASK`, in server order, without duplicates. May contain `*`. */
 | {
    readonly kind: "union";
    readonly names: readonly string[];
}
/** Exactly `*`: connects to anything. */
 | {
    readonly kind: "any";
};
/** Fields shared by every input. */
type InputBase = {
    /** The input's key on the wire, exactly as on the server. Never split on dots. */
    readonly name: string;
    /** Whether the backend requires a value. */
    readonly required: boolean;
    readonly tooltip?: string;
    /** Evaluated only when the node asks for it. */
    readonly lazy: boolean;
    /** The node receives the link itself instead of its value. */
    readonly rawLink: boolean;
};
/** Fields shared by inputs the frontend shows as a widget. */
type WidgetBase = InputBase & {
    /** The type a link into this input must match, and the type the backend coerces by. */
    readonly type: SocketType;
    /**
     * The server asks for no socket. Frontend 1.55.17 ignores it (every widget gets a slot), so it
     * only makes the generated types refuse links, which is stricter than the frontend and backend.
     */
    readonly socketless: boolean;
    /**
     * The frontend widget the server names, when it differs from the one `kind` implies: a pack's
     * editor for a STRING, or a core widget on a primitive type, whose coercion decides the kind.
     */
    readonly widgetType?: string;
};
/** Takes links only: a custom type, `*`, a comma union, or a widget type with `forceInput`. */
export type SocketInput = InputBase & {
    readonly kind: "socket";
    readonly type: SocketType;
    /**
     * The server marks it `forceInput`: files from frontends before 1.16 saved a placeholder value
     * for it in `widgets_values`, which the frontend drops on load.
     */
    readonly forceInput?: true;
    /** `control_after_generate` as the server states it, for a `forceInput` input. */
    readonly control?: boolean | string;
};
/** An INT widget. */
export type IntInput = WidgetBase & {
    readonly kind: "int";
    readonly default?: IntValue;
    readonly min?: IntValue;
    readonly max?: IntValue;
    readonly step?: number;
    /**
     * `control_after_generate` as the server states it. The frontend adds a control widget
     * (randomize, fixed, …) when it is truthy, named after it when it is a string, and also when
     * it is absent and the input is named `seed` or `noise_seed`. The control is saved in UI
     * workflows but never sent.
     */
    readonly control?: boolean | string;
};
/** A FLOAT widget. Bounds the server wrote beyond 2^53 are rounded inward. */
export type FloatInput = WidgetBase & {
    readonly kind: "float";
    readonly default?: number;
    readonly min?: number;
    readonly max?: number;
    readonly step?: number;
    /** Rounding precision the frontend applies, or `false` for none. */
    readonly round?: number | false;
    /** `control_after_generate` as the server states it; see {@link IntInput.control}. */
    readonly control?: boolean | string;
};
/** A STRING widget. */
export type StringInput = WidgetBase & {
    readonly kind: "string";
    readonly default?: string;
    readonly multiline: boolean;
    /** The frontend rewrites `{a|b}` and comments in this input; tscomfy doesn't. */
    readonly dynamicPrompts: boolean;
};
/** A BOOLEAN widget. */
export type BooleanInput = WidgetBase & {
    readonly kind: "boolean";
    readonly default?: boolean;
};
/** An upload flag of a combo: `image_upload`, `video_upload`, … without `_upload`. */
export type ComboUpload = "image" | "video" | "animated_image" | "audio" | "file";
/** A combo, from either the legacy `[[…]]` form or the `["COMBO", {options}]` form. */
export type ComboInput = WidgetBase & {
    readonly kind: "combo";
    /**
     * The backend matches links into the legacy form against the option list itself, and into
     * the `COMBO` form by `type`. A legacy combo's `type` is `COMBO`, as the frontend sees it.
     */
    readonly form: "legacy" | "combo";
    /** Server order. May be empty (no model files installed, or a remote combo). */
    readonly options: readonly ComboOption[];
    /**
     * What the frontend sends when no value is given: `[]` for a multi-select widget, whatever the
     * server states; otherwise the server's default, with the first option in place of a falsy one.
     * Absent when the server states none, and for a `forceInput` combo, which has no widget.
     */
    readonly default?: ComboOption | readonly ComboOption[];
    /** The backend validates a list value element by element. */
    readonly multiselect: boolean;
    /** Options are files the user can upload, so they change at runtime. */
    readonly upload: boolean;
    /** Which upload flags the server sets (`image_upload` → `image`), when any. */
    readonly uploads?: readonly ComboUpload[];
    /** Options are fetched by the frontend at runtime. */
    readonly remote: boolean;
    /**
     * A remote combo whose `remote.refresh_button` asks the frontend for a refresh button, which
     * it saves after the combo.
     */
    readonly refreshButton?: true;
    /** `control_after_generate` as the server states it; see {@link IntInput.control}. */
    readonly control?: boolean | string;
    /**
     * A legacy combo with `forceInput`: the frontend draws a socket and no widget, so it sends a
     * link or nothing, and it has no default.
     */
    readonly forceInput?: true;
};
/**
 * A frontend widget whose value is arbitrary JSON (COLOR, BOUNDING_BOX, CURVE, …). The backend
 * coerces by `type`, not by the widget.
 */
export type ValueInput = WidgetBase & {
    readonly kind: "value";
    /** The frontend widget that edits the value. */
    readonly widget: string;
    /** What the frontend sends when no value is given: the server's default, for most widgets. */
    readonly default?: JsonValue;
};
/** `COMFY_AUTOGROW_V3`: a growing list of sockets built from one template input. */
export type AutogrowInput = InputBase & {
    readonly kind: "autogrow";
    /** `prefix`: slots `<prefix>0`, `<prefix>1`, …; `names`: a fixed list of slot names. */
    readonly form: "prefix" | "names";
    /** Every slot name, in order. Wire keys are `<name>.<slot>`. */
    readonly slots: readonly string[];
    /** Slots before this index are required when the template is. */
    readonly min: number;
    /** The input every slot copies. Always link-only: the backend forces it to a socket. */
    readonly template: InputDef;
};
/** One choice of a {@link DynamicComboInput}. */
export type DynamicComboOption = {
    readonly key: string;
    /** Inputs that exist only while this option is selected. Wire keys are `<name>.<child>`. */
    readonly inputs: readonly InputDef[];
};
/** `COMFY_DYNAMICCOMBO_V3`: a combo whose selected option adds its own inputs. */
export type DynamicComboInput = InputBase & {
    readonly kind: "dynamicCombo";
    readonly options: readonly DynamicComboOption[];
};
/** `COMFY_DYNAMICGROUP_V3`: rows of widget fields. Upstream marks it unstable. */
export type DynamicGroupInput = InputBase & {
    readonly kind: "dynamicGroup";
    /** The fields of one row. Wire keys are `<name>.<row>.<field>`. */
    readonly fields: readonly InputDef[];
    readonly min: number;
    readonly max: number;
    readonly groupName?: string;
};
/** `COMFY_MATCHTYPE_V3`: takes any allowed type, shared with other inputs and outputs. */
export type MatchTypeInput = InputBase & {
    readonly kind: "matchType";
    readonly templateId: string;
    readonly allowed: SocketType;
};
/**
 * An input spec the normalizer couldn't interpret. It is kept, with a diagnostic, so later stages
 * can degrade gracefully instead of losing the input.
 */
export type UnknownInput = InputBase & {
    readonly kind: "unknown";
    /** The spec exactly as the server sent it. */
    readonly spec: JsonValue;
};
/** One input of a node. */
export type InputDef = SocketInput | IntInput | FloatInput | StringInput | BooleanInput | ComboInput | ValueInput | AutogrowInput | DynamicComboInput | DynamicGroupInput | MatchTypeInput | UnknownInput;
/** The type of an output slot. */
export type OutputType = SocketType
/** An option list, which may connect to a `COMBO`-form combo input. */
 | {
    readonly kind: "combo";
    readonly options: readonly ComboOption[];
};
/** One output slot of a node. */
export type OutputDef = {
    /** The slot index used in links. Outputs are never filtered, so this is also the array index. */
    readonly index: number;
    /** Display name. Not unique: two outputs may share a name. */
    readonly name: string;
    readonly type: OutputType;
    /** The node emits a list; ComfyUI maps it over a non-list input downstream. */
    readonly isList: boolean;
    readonly tooltip?: string;
    /** The MatchType template this output follows; its `type` is then `any`. */
    readonly matchTemplate?: string;
};
/** One node class. */
export type NodeDef = {
    /** The key used as `class_type` in prompts, exactly as on the server. */
    readonly classType: string;
    readonly displayName: string;
    readonly category: string;
    /** The Python module that defines the node, which identifies its pack. */
    readonly pythonModule: string;
    readonly description?: string;
    /** The node produces results (save, preview), so the backend executes it. */
    readonly outputNode: boolean;
    /** The node receives every input as a whole list. */
    readonly isInputList: boolean;
    readonly deprecated: boolean;
    readonly experimental: boolean;
    /** In `input_order`; hidden inputs are dropped. */
    readonly inputs: readonly InputDef[];
    readonly outputs: readonly OutputDef[];
};
/** The node definitions of one ComfyUI server, normalized. */
export type Schema = {
    readonly formatVersion: 1;
    /** The ComfyUI version, if the caller supplied it. Metadata only: not part of the hash. */
    readonly comfyVersion?: string;
    /**
     * Keyed by class type and built in code-unit order of class type. JavaScript enumerates
     * integer-like keys first, in numeric order, so a class named "10" would list before "A".
     */
    readonly nodes: {
        readonly [classType: string]: NodeDef;
    };
};
export {};
```
